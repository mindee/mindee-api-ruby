def to_s
  [
    ":ID: #{@id}",
    ":Model ID: #{@model_id}",
    ":Filename: #{@filename}",
    ":Created At: #{@created_at}",
    ":Total Matches: #{@total_matches}",
    ":Last Match At: #{@last_match_at}",
    ":Status: #{@status}",
  ].join("\n")
end
